const intent = {
  robot: "mini-arm",
  action: "joint.move",
  bounds: {
    max_velocity: 0.5,
    horizon_ms: 500
  },
  authority: "scoped-capability",
  evidence: "action-bound"
};
// Evidence tied to the authorized action

  intent_digest: "sha256:…",
  capability_digest: "sha256:…",

    scope: "joint.position",

  },

  signature: "…"
